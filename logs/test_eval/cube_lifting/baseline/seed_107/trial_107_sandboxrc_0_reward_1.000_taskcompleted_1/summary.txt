def rot_to_quat_wxyz(Rm):
    t = np.trace(Rm)
    if t > 0:
        s = np.sqrt(t + 1.0) * 2
        w = 0.25 * s
        x = (Rm[2,1] - Rm[1,2]) / s
        y = (Rm[0,2] - Rm[2,0]) / s
        z = (Rm[1,0] - Rm[0,1]) / s
    elif Rm[0,0] > Rm[1,1] and Rm[0,0] > Rm[2,2]:
        s = np.sqrt(1.0 + Rm[0,0] - Rm[1,1] - Rm[2,2]) * 2
        w = (Rm[2,1] - Rm[1,2]) / s
        x = 0.25 * s
        y = (Rm[0,1] + Rm[1,0]) / s
        z = (Rm[0,2] + Rm[2,0]) / s
    elif Rm[1,1] > Rm[2,2]:
        s = np.sqrt(1.0 + Rm[1,1] - Rm[0,0] - Rm[2,2]) * 2
        w = (Rm[0,2] - Rm[2,0]) / s
        x = (Rm[0,1] + Rm[1,0]) / s
        y = 0.25 * s
        z = (Rm[1,2] + Rm[2,1]) / s
    else:
        s = np.sqrt(1.0 + Rm[2,2] - Rm[0,0] - Rm[1,1]) * 2
        w = (Rm[1,0] - Rm[0,1]) / s
        x = (Rm[0,2] + Rm[2,0]) / s
        y = (Rm[1,2] + Rm[2,1]) / s
        z = 0.25 * s
    q = np.array([w, x, y, z])
    return q / np.linalg.norm(q)
