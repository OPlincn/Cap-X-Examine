def mask_to_world_points(mask, depth, K, cam_ext):
    ys, xs = np.where(mask)
    zs = depth[ys, xs]
    valid = zs > 0
    xs, ys, zs = xs[valid], ys[valid], zs[valid]
    fx, fy = K[0, 0], K[1, 1]
    cx, cy = K[0, 2], K[1, 2]
    x_cam = (xs - cx) * zs / fx

    red_pts = mask_to_world_points(red_mask, depth, K, cam_ext)
    green_pts = mask_to_world_points(green_mask, depth, K, cam_ext)
    red_obb = get_oriented_bounding_box_from_3d_points(red_pts)
    green_obb = get_oriented_bounding_box_from_3d_points(green_pts)

    # Step 2: grasp red cube
    red_seg = red_mask.astype(np.int32)
    grasp_poses, grasp_scores = plan_grasp(depth=depth, intrinsics=K, segmentation=red_seg)
    best_idx = grasp_scores.argmax()
    best_T_world = cam_ext @ grasp_poses[best_idx]
    red_grasp_pos = best_T_world[:3, 3]
    red_grasp_quat = rot_to_quat(best_T_world[:3, :3])

    open_gripper()
    pre_grasp_pos = red_grasp_pos.copy()
    pre_grasp_pos[2] += 0.10
    joints = solve_ik(pre_grasp_pos, red_grasp_quat)
    move_to_joints(joints)
    joints = solve_ik(red_grasp_pos, red_grasp_quat)
    move_to_joints(joints)
    close_gripper()

    lift_pos = red_grasp_pos.copy()
    lift_pos[2] += 0.20
    joints = solve_ik(lift_pos, red_grasp_quat)
    move_to_joints(joints)

    # Step 3: place red on top of green
    green_top_z = green_obb["center"][2] + green_obb["extent"][2] / 2.0
    red_bottom_offset = red_obb["extent"][2] / 2.0

    place_pos = np.array([
        green_obb["center"][0],
        green_obb["center"][1],
        green_top_z + red_bottom_offset + 0.01
    ])

    above_place = place_pos.copy()
    above_place[2] += 0.10
    joints = solve_ik(above_place, red_grasp_quat)
    move_to_joints(joints)
    joints = solve_ik(place_pos, red_grasp_quat)
    move_to_joints(joints)

    open_gripper()

    clear_pos = place_pos.copy()
    clear_pos[2] += 0.15
    joints = solve_ik(clear_pos, red_grasp_quat)
    move_to_joints(joints)
