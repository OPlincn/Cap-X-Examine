```python
import numpy as np

obs = get_observation()
cam = obs["robot0_robotview"]
rgb = cam["images"]["rgb"]
depth = cam["images"]["depth"]
K = cam["intrinsics"]
cam_pose = cam["pose_mat"]

masks = segment_sam3_text_prompt(rgb, text_prompt="brown spill")
best = max(masks, key=lambda d: d["score"])
mask = best["mask"]

# Backproject masked pixels to 3D in camera frame, then to world
ys, xs = np.where(mask)
zs = depth[ys, xs]
valid = zs > 0
xs, ys, zs = xs[valid], ys[valid], zs[valid]

fx, fy = K[0, 0], K[1, 1]
cx, cy = K[0, 2], K[1, 2]
Xc = (xs - cx) * zs / fx
Yc = (ys - cy) * zs / fy
Zc = zs
pts_cam = np.stack([Xc, Yc, Zc], axis=1)

# camera pose_mat maps camera -> world
R = cam_pose[:3, :3]
t = cam_pose[:3, 3]
pts_world = (R @ pts_cam.T).T + t

print("world x range:", pts_world[:,0].min(), pts_world[:,0].max())
print("world y range:", pts_world[:,1].min(), pts_world[:,1].max())
print("world z range:", pts_world[:,2].min(), pts_world[:,2].max())
print("num points:", len(pts_world))
```

import numpy as np

# Recompute extents (state may have changed slightly)
obs = get_observation()
cam = obs["robot0_robotview"]
rgb = cam["images"]["rgb"]
depth = cam["images"]["depth"]
K = cam["intrinsics"]
cam_pose = cam["pose_mat"]

masks = segment_sam3_text_prompt(rgb, text_prompt="brown spill")
best = max(masks, key=lambda d: d["score"])
mask = best["mask"]

ys, xs = np.where(mask)
zs = depth[ys, xs]
valid = zs > 0
xs, ys, zs = xs[valid], ys[valid], zs[valid]
fx, fy = K[0, 0], K[1, 1]
cx, cy = K[0, 2], K[1, 2]
Xc = (xs - cx) * zs / fx
Yc = (ys - cy) * zs / fy
pts_cam = np.stack([Xc, Yc, zs], axis=1)
R = cam_pose[:3, :3]
t = cam_pose[:3, 3]
pts_world = (R @ pts_cam.T).T + t

xmin, xmax = pts_world[:,0].min(), pts_world[:,0].max()
ymin, ymax = pts_world[:,1].min(), pts_world[:,1].max()

quat = np.array([0.0, 0.0, 1.0, 0.0])

# Small margin inside extents to keep motions modest
mx = 0.01
my = 0.01
x0, x1 = xmin + mx, xmax - mx
y0, y1 = ymin + my, ymax - my
if x1 <= x0:
    x0, x1 = xmin, xmax
if y1 <= y0:
    y0, y1 = ymin, ymax

# Serpentine wiping path at z=0 with small steps
n_x = 4
n_y = 4
xs_path = np.linspace(x0, x1, n_x)
ys_path = np.linspace(y0, y1, n_y)

waypoints = []
for j, yy in enumerate(ys_path):
    xseq = xs_path if j % 2 == 0 else xs_path[::-1]
    for xx in xseq:
        waypoints.append(np.array([xx, yy, 0.0]))

# Execute wiping waypoints
for wp in waypoints:
    joints = solve_ik(wp, quat)
    move_to_joints(joints)

# Lift up after wiping
lift = np.array([(x0+x1)/2.0, (y0+y1)/2.0, 0.15])
joints = solve_ik(lift, quat)
move_to_joints(joints)
print("wiping complete")
