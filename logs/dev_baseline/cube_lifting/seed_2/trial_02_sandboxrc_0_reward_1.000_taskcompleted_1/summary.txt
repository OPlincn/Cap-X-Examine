import numpy as np
obs = get_observation()
cam = obs["robot0_robotview"]
rgb = cam["images"]["rgb"]
depth = cam["images"]["depth"]
K = cam["intrinsics"]

masks = segment_sam3_text_prompt(rgb, text_prompt="red cube")
best = max(masks, key=lambda d: d["score"])
mask = best["mask"].astype(np.int32)

# Compute cube center in world frame from mask + depth
ys, xs = np.where(mask > 0)
zs = depth[ys, xs]
valid = zs > 0
xs, ys, zs = xs[valid], ys[valid], zs[valid]
fx, fy = K[0,0], K[1,1]
cx, cy = K[0,2], K[1,2]
X = (xs - cx) * zs / fx
Y = (ys - cy) * zs / fy
Z = zs
pts_cam = np.stack([X, Y, Z], axis=1)
pts_cam_h = np.concatenate([pts_cam, np.ones((pts_cam.shape[0],1))], axis=1)
pts_world = (ext @ pts_cam_h.T).T[:, :3]

center = pts_world.mean(axis=0)
print("cube center world", center)

# Top-down grasp orientation: gripper pointing down (z axis of hand = -world z)
# Franka hand default: approach along +z of hand. We want hand z pointing down.
# Quaternion for 180 deg rotation about x: w=0, x=1, y=0, z=0
quat = np.array([0.0, 1.0, 0.0, 0.0])

# Pre-grasp above cube
pre = center.copy()
pre[2] += 0.15

move_to_joints(joints)

# Move down to grasp at cube center height
